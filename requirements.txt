# TopRate backend dependencies.

# scikit-learn is PINNED EXACTLY and deliberately. The WPR projection model is
# a HistGradientBoostingRegressor saved with joblib. sklearn moved its internal
# `_loss` module between versions, so a model pickled under one version fails to
# load under another with "No module named '_loss'". Pin sklearn here and
# install from this file in BOTH your local env and the GitHub Action so the
# training environment and the serving environment are identical and the saved
# model always loads.
#
# IF YOU UPGRADE sklearn: you MUST retrain in the same commit, then bump the pin
#   python wpr_projection.py --retrain --jobs -1
#
# The other packages use >= floors (not exact pins) since they don't affect
# model loading; only sklearn does. If you want fully reproducible installs,
# replace the >= with == and your actual installed versions (pip freeze).
scikit-learn==1.8.0
joblib>=1.3
numpy>=1.26
pandas>=2.0
requests>=2.31
urllib3>=2.0
